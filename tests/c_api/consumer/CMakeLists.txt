cmake_minimum_required(VERSION 3.28)
project(rollback_lab_sdk_consumer LANGUAGES C CXX)
enable_testing()
find_package(rollback_lab 0.2.0 EXACT CONFIG REQUIRED)
add_executable(c_consumer ../c_consumer.c)
target_link_libraries(c_consumer PRIVATE rollback_lab::rollback_lab_c)
set_target_properties(c_consumer PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED ON C_EXTENSIONS OFF)
add_executable(cpp_consumer cpp_consumer.cpp)
target_link_libraries(cpp_consumer PRIVATE rollback_lab::rollback_lab_c rollback_lab::rollback_lab)
if(MSVC)
    target_compile_options(c_consumer PRIVATE /W4 /WX)
    target_compile_options(cpp_consumer PRIVATE /W4 /WX /permissive- /EHsc)
else()
    target_compile_options(c_consumer PRIVATE -Wall -Wextra -Wpedantic -Werror)
    target_compile_options(cpp_consumer PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
add_test(NAME installed_c11_consumer COMMAND c_consumer ${SDK_EXPECTED_SHA})
add_test(NAME installed_cpp_consumer COMMAND cpp_consumer ${SDK_EXPECTED_SHA})
